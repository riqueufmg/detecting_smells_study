    /**
     * Performs generic type inference on a resolved method usage by analyzing the relationship
     * between formal parameter types and actual parameter types provided at the call site.
     * Handles both regular methods and varargs methods with appropriate constraint collection.
     *
     * @param methodUsage the initially resolved method usage
     * @param parameterTypes the actual parameter types from the call site
     * @return an Optional containing the method usage with inferred generic types, or empty if inference fails
     */
    private Optional<MethodUsage> performTypeInference(MethodUsage methodUsage, List<ResolvedType> parameterTypes) {
        try {
            InferenceContext inferenceContext = new InferenceContext(typeSolver);
            List<ResolvedType> constraintPairs = collectTypeConstraints(methodUsage, parameterTypes, inferenceContext);
            return resolveInferredTypes(methodUsage, constraintPairs, inferenceContext);
        } catch (ConflictingGenericTypesException e) {
            return Optional.empty();
        }
    }

    /**

     *
     * @param methodUsage the method usage to analyze
     * @param parameterTypes the actual parameter types from the call site
     * @param inferenceContext the inference context for collecting type constraints
     * @return a list of type constraints that will be used for generic type resolution
     * @throws IllegalArgumentException if parameter counts are incompatible
     */
    private List<ResolvedType> collectTypeConstraints(
            MethodUsage methodUsage, List<ResolvedType> parameterTypes, InferenceContext inferenceContext) {

        boolean isVarArgs = methodUsage.getDeclaration().hasVariadicParameter();
        int formalParamCount = methodUsage.getNoParams();
        int actualParamCount = parameterTypes.size();

        validateParameterCount(isVarArgs, formalParamCount, actualParamCount);

        if (isVarArgs) {
            return collectVarArgsConstraints(methodUsage, parameterTypes, inferenceContext, formalParamCount);
        } else {
            return collectRegularConstraints(methodUsage, parameterTypes, inferenceContext);
        }
    }

   256	    /**
   257	     * Collects type constraints for varargs methods. This involves two phases:
   258	     * 1. Regular parameters: handled like non-varargs methods (one-to-one mapping)
   259	     * 2. Varargs parameter: handled specially based on how arguments are passed
   260	     *
   261	     * The varargs parameter can receive arguments in two ways:
   262	     * - Direct array passing: method(array) - constraint between array types
   263	     * - Individual elements: method(elem1, elem2, ...) - constraints between component type and each element
   264	     *
   265	     * @param methodUsage the varargs method usage to analyze
   266	     * @param parameterTypes the actual parameter types from the call site
   267	     * @param inferenceContext the inference context for collecting constraints
   268	     * @param formalParamCount the total number of formal parameters (including varargs)
   269	     * @return a list of type constraints covering both regular and varargs parameters
   270	     */
   271	    private List<ResolvedType> collectVarArgsConstraints(
   272	            MethodUsage methodUsage,
   273	            List<ResolvedType> parameterTypes,
   274	            InferenceContext inferenceContext,
   275	            int formalParamCount) {
   276	        List<ResolvedType> constraints = new LinkedList<>();
   277	        int regularParamCount = formalParamCount - 1;
   278	
   279	        // Process regular parameters
   280	        for (int i = 0; i < regularParamCount; i++) {
   281	            ResolvedType formalType = methodUsage.getParamType(i);
   282	            ResolvedType actualType = parameterTypes.get(i);
   283	            constraints.add(inferenceContext.addPair(formalType, actualType));
   284	        }

   287	        if (formalParamCount > 0) {

   289	            processVarArgsParameter(varargsParamType, parameterTypes, regularParamCount, inferenceContext, constraints);

   295	    /**
   296	     * Processes the varargs parameter by determining how arguments are passed and creating
   297	     * appropriate type constraints. Handles two scenarios:
   298	     *
   299	     * 1. Direct array passing: When exactly one argument is passed to varargs and it's an array,
   300	     *    creates a constraint between the formal array type and the actual array type.
   301	     *    Example: method(String[] args) called as method(stringArray)
   302	     *
   303	     * 2. Individual element passing: When multiple arguments are passed to varargs,
   304	     *    creates constraints between the array's component type and each individual argument.
   305	     *    Example: method(String... args) called as method("a", "b", "c")
   306	     *
   307	     * @param varargsParamType the formal type of the varargs parameter (must be an array type)
   308	     * @param parameterTypes all actual parameter types from the call site
   309	     * @param regularParamCount the number of regular (non-varargs) parameters
   310	     * @param inferenceContext the inference context for collecting constraints
   311	     * @param constraints the constraint list to add new constraints to
   312	     * @throws IllegalStateException if the varargs parameter is not an array type
   313	     */
   314	    private void processVarArgsParameter(
   315	            ResolvedType varargsParamType,
   316	            List<ResolvedType> parameterTypes,
   317	            int regularParamCount,
   318	            InferenceContext inferenceContext,
   319	            List<ResolvedType> constraints) {
   320	        if (!varargsParamType.isArray()) {
   321	            throw new IllegalStateException("Varargs parameter is not an array type");
   322	        }
   323	
   324	        ResolvedType componentType = varargsParamType.asArrayType().getComponentType();
   325	        int actualParamCount = parameterTypes.size();
   326	
   327	        if (isDirectArrayPassing(parameterTypes, regularParamCount, actualParamCount)) {
   328	            // Direct array passing: method(array)
   329	            ResolvedType actualArrayType = parameterTypes.get(regularParamCount);
   330	            constraints.add(inferenceContext.addPair(varargsParamType, actualArrayType));
   331	        } else {
   332	            // Individual elements passing: method(elem1, elem2, ...)
   333	            for (int i = regularParamCount; i < actualParamCount; i++) {
   334	                ResolvedType actualType = parameterTypes.get(i);
   335	                constraints.add(inferenceContext.addPair(componentType, actualType));
   336	            }
   337	        }
   338	    }

   341	     * Determines whether arguments are being passed directly as an array to the varargs parameter.
   342	     * This happens when there is exactly one argument for the varargs parameter and that argument
   343	     * is an array type. This is a special case that requires different constraint handling.
   344	     *
   345	     * @param parameterTypes all actual parameter types from the call site
   346	     * @param regularParamCount the number of regular (non-varargs) parameters
   347	     * @param actualParamCount the total number of actual parameters
   348	     * @return true if a single array is being passed directly to varargs, false otherwise
   349	     */
   350	    private boolean isDirectArrayPassing(
   351	            List<ResolvedType> parameterTypes, int regularParamCount, int actualParamCount) {
   352	        return actualParamCount == regularParamCount + 1
   353	                && parameterTypes.get(regularParamCount).isArray();

   357	     * Applies the inferred generic types to the method usage by resolving all collected constraints
   358	     * and updating both parameter types and return type with their concrete resolved types.
   359	     * This is the final step that produces a fully resolved MethodUsage with no remaining generic placeholders.
   360	     *
   361	     * @param methodUsage the method usage to update with resolved types
   362	     * @param constraints the collected type constraints from parameter analysis
   363	     * @param inferenceContext the inference context containing all type relationships
   364	     * @return an Optional containing the fully resolved MethodUsage
   365	     * @throws ConflictingGenericTypesException if type constraints are contradictory
   366	     */
   367	    private Optional<MethodUsage> resolveInferredTypes(
   368	            MethodUsage methodUsage, List<ResolvedType> constraints, InferenceContext inferenceContext) {
   369	        // Resolve return type
   370	        ResolvedType returnType = inferenceContext.addSingle(methodUsage.returnType());
   371	
   372	        // Apply resolved parameter types
   373	        for (int i = 0; i < constraints.size() && i < methodUsage.getNoParams(); i++) {

   375	            methodUsage = methodUsage.replaceParamType(i, resolvedParamType);
   376	        }

   378	        // Apply resolved return type
   379	        methodUsage = methodUsage.replaceReturnType(inferenceContext.resolve(returnType));
   380	
   381	        return Optional.of(methodUsage);
