   150	    public SymbolReference<ResolvedConstructorDeclaration> solve(

   152	        // Constructor invocation must exist within a class (not interface).

   154	                explicitConstructorInvocationStmt.findAncestor(ClassOrInterfaceDeclaration.class);

   156	            return unsolved();

   159	        ClassOrInterfaceDeclaration classOrInterfaceNode = optAncestorClassOrInterfaceNode.get();
   160	        ResolvedReferenceTypeDeclaration resolvedClassNode = classOrInterfaceNode.resolve();
   161	        if (!resolvedClassNode.isClass()) {
   162	            throw new IllegalStateException(
   163	                    "Expected to be a class -- cannot call this() or super() within an interface.");
   164	        }

   166	        ResolvedTypeDeclaration typeDecl = null;
   167	        if (explicitConstructorInvocationStmt.isThis()) {
   168	            // this()

   173	                    resolvedClassNode.asClass().getSuperClass();

   175	                typeDecl = superClass.get().getTypeDeclaration().get();

   257	    private void solveArguments(
   258	            Node node,
   259	            NodeList<Expression> args,
   260	            boolean solveLambdas,
   261	            List<ResolvedType> argumentTypes,
   262	            List<LambdaArgumentTypePlaceholder> placeholders) {
   263	        int i = 0;
   264	        for (Expression parameterValue : args) {
   265	            while (parameterValue instanceof EnclosedExpr) {
   266	                parameterValue = ((EnclosedExpr) parameterValue).getInner();
   267	            }
   268	            // In order to resolve a call with a lambda expr as an argument, the functional interface implemented
   269	            // by that lambda must be determined. This is done by collecting the candidate functional methods in
   270	            // scope and then excluding non-applicable methods if any of the following hold:
   271	            //   1. The lambda and the candidate method do not have the same number of arguments/parameters
   272	            //   2. The lambda has an explicit empty/void return statement `return;` in the body block, but the
   273	            //      candidate method has a non-void return type.
   274	            //   3. The lambda has an explicit non-void return statement, e.g. `return x;`, but the candidate
   275	            //      method has a void return type.
   276	            // For JavaParser to be able to perform the same filtering, we need to keep track of the number of
   277	            // arguments to the lambda, as well as whether there is an explicit void/non-void return statement,
   278	            // so this information is added to the `LambdaArgumentTypePlaceholder` in the block below.
   279	            if (parameterValue.isLambdaExpr()) {
   280	                LambdaExpr lambdaExpr = parameterValue.asLambdaExpr();
   281	                Optional<Boolean> bodyBlockHasExplicitNonVoidReturn;
   282	                if (!lambdaExpr.getBody().isBlockStmt()) {
   283	                    bodyBlockHasExplicitNonVoidReturn = Optional.empty();
   284	                } else {
   285	                    Optional<ReturnStmt> explicitReturn = lambdaExpr.getBody().findFirst(ReturnStmt.class);
   286	                    if (explicitReturn.isPresent()) {
   287	                        bodyBlockHasExplicitNonVoidReturn =
   288	                                Optional.of(explicitReturn.get().getExpression().isPresent());
   289	                    } else {
   290	                        bodyBlockHasExplicitNonVoidReturn = Optional.of(false);
   291	                    }
   292	                }
   293	                LambdaArgumentTypePlaceholder placeholder = new LambdaArgumentTypePlaceholder(
   294	                        i, lambdaExpr.getParameters().size(), bodyBlockHasExplicitNonVoidReturn);
   295	                argumentTypes.add(placeholder);
   296	                placeholders.add(placeholder);
   297	            } else if (parameterValue.isMethodReferenceExpr()) {
   298	                LambdaArgumentTypePlaceholder placeholder = new LambdaArgumentTypePlaceholder(i);
   299	                argumentTypes.add(placeholder);
   300	                placeholders.add(placeholder);
   301	            } else {
   302	                try {
   303	                    argumentTypes.add(JavaParserFacade.get(typeSolver).getType(parameterValue, solveLambdas));
   304	                } catch (Exception e) {
   305	                    throw failureHandler.handle(
   306	                            e,
   307	                            String.format(
   308	                                    "Unable to calculate the type of a parameter of a method call. Method call: %s, Parameter: %s",
   309	                                    node, parameterValue));
   310	                }
   311	            }
   312	            i++;
   313	        }
   314	    }

    68	    private static boolean isApplicable(
    69	            ResolvedConstructorDeclaration constructor,
    70	            List<ResolvedType> argumentsTypes,
    71	            TypeSolver typeSolver,
    72	            boolean withWildcardTolerance) {
    73	        if (constructor.hasVariadicParameter()) {
    74	            int pos = constructor.getNumberOfParams() - 1;
    75	            if (constructor.getNumberOfParams() == argumentsTypes.size()) {
    76	                // check if the last value is directly assignable as an array
    77	                ResolvedType expectedType = constructor.getLastParam().getType();
    78	                ResolvedType actualType = argumentsTypes.get(pos);
    79	                if (!expectedType.isAssignableBy(actualType)) {
    80	                    for (ResolvedTypeParameterDeclaration tp : constructor.getTypeParameters()) {
    81	                        expectedType = MethodResolutionLogic.replaceTypeParam(expectedType, tp, typeSolver);
    82	                    }
    83	                    if (!expectedType.isAssignableBy(actualType)) {
    84	                        if (actualType.isArray()
    85	                                && expectedType.isAssignableBy(
    86	                                        actualType.asArrayType().getComponentType())) {
    87	                            argumentsTypes.set(pos, actualType.asArrayType().getComponentType());
    88	                        } else {
    89	                            argumentsTypes = groupVariadicParamValues(
    90	                                    argumentsTypes,
    91	                                    pos,
    92	                                    constructor.getLastParam().getType());
    93	                        }
    94	                    }
    95	                }
    96	                // else it is already assignable, nothing to do
    97	            } else {
    98	                if (pos > argumentsTypes.size()) {
    99	                    return false;
   100	                }
   101	                argumentsTypes = groupVariadicParamValues(
   102	                        argumentsTypes, pos, constructor.getLastParam().getType());
   103	            }
   104	        }
   105	        if (constructor.getNumberOfParams() != argumentsTypes.size()) {
   106	            return false;
   107	        }
   108	        Map<String, ResolvedType> matchedParameters = new HashMap<>();
   109	        boolean needForWildCardTolerance = false;
   110	        for (int i = 0; i < constructor.getNumberOfParams(); i++) {
   111	            ResolvedType expectedType = constructor.getParam(i).getType();
   112	            ResolvedType actualType = argumentsTypes.get(i);
   113	            if ((expectedType.isTypeVariable() && !(expectedType.isWildcard()))
   114	                    && expectedType.asTypeParameter().declaredOnMethod()) {
   115	                matchedParameters.put(expectedType.asTypeParameter().getName(), actualType);
   116	                continue;
   117	            }
   118	            boolean isAssignableWithoutSubstitution = expectedType.isAssignableBy(actualType)
   119	                    || (constructor.getParam(i).isVariadic()
   120	                            && new ResolvedArrayType(expectedType).isAssignableBy(actualType));
   121	            if (!isAssignableWithoutSubstitution && expectedType.isReferenceType() && actualType.isReferenceType()) {
   122	                isAssignableWithoutSubstitution = MethodResolutionLogic.isAssignableMatchTypeParameters(
   123	                        expectedType.asReferenceType(), actualType.asReferenceType(), matchedParameters);
   124	            }
   125	            if (!isAssignableWithoutSubstitution) {
   126	                for (ResolvedTypeParameterDeclaration tp : constructor.getTypeParameters()) {
   127	                    expectedType = MethodResolutionLogic.replaceTypeParam(expectedType, tp, typeSolver);
   128	                }

   130	                        constructor.declaringType().getTypeParameters()) {
   131	                    expectedType = MethodResolutionLogic.replaceTypeParam(expectedType, tp, typeSolver);
   132	                }

   134	                    if (actualType.isWildcard() && withWildcardTolerance && !expectedType.isPrimitive()) {
   135	                        needForWildCardTolerance = true;
   136	                        continue;
   137	                    }

   139	                        if (new ResolvedArrayType(expectedType).isAssignableBy(actualType)) {
   140	                            continue;
   141	                        }
   142	                    }
   143	                    return false;
   144	                }
   145	            }
   146	        }
   147	        return !withWildcardTolerance || needForWildCardTolerance;

   168	    public static SymbolReference<ResolvedConstructorDeclaration> findMostApplicable(
   169	            List<ResolvedConstructorDeclaration> constructors,
   170	            List<ResolvedType> argumentsTypes,
   171	            TypeSolver typeSolver,
   172	            boolean wildcardTolerance) {
   173	        List<ResolvedConstructorDeclaration> applicableConstructors = constructors.stream()
   174	                .filter((m) -> isApplicable(m, argumentsTypes, typeSolver, wildcardTolerance))
   175	                .collect(Collectors.toList());
   176	        if (applicableConstructors.isEmpty()) {
   177	            return SymbolReference.unsolved();
   178	        }
   179	        if (applicableConstructors.size() == 1) {
   180	            return SymbolReference.solved(applicableConstructors.get(0));
   181	        }
   182	        ResolvedConstructorDeclaration winningCandidate = applicableConstructors.get(0);
   183	        ResolvedConstructorDeclaration other = null;
   184	        boolean possibleAmbiguity = false;
   185	        for (int i = 1; i < applicableConstructors.size(); i++) {
   186	            other = applicableConstructors.get(i);
   187	            if (MethodResolutionLogic.isMoreSpecific(winningCandidate, other, argumentsTypes)) {
   188	                possibleAmbiguity = false;
   189	            } else if (MethodResolutionLogic.isMoreSpecific(other, winningCandidate, argumentsTypes)) {
   190	                possibleAmbiguity = false;
   191	                winningCandidate = other;
   192	            } else {
   193	                if (winningCandidate
   194	                        .declaringType()
   195	                        .getQualifiedName()
   196	                        .equals(other.declaringType().getQualifiedName())) {
   197	                    possibleAmbiguity = true;
   198	                } else {
   199	                    // we expect the methods to be ordered such that inherited methods are later in the list
   200	                }
   201	            }
   202	            if (possibleAmbiguity) {
   203	                // pick the first exact match if it exists
   204	                if (!MethodResolutionLogic.isExactMatch(winningCandidate, argumentsTypes)) {
   205	                    if (MethodResolutionLogic.isExactMatch(other, argumentsTypes)) {
   206	                        winningCandidate = other;
   207	                    } else {
   208	                        throw new MethodAmbiguityException(
   209	                                "Ambiguous constructor call: cannot find a most applicable constructor: "
   210	                                        + winningCandidate + ", " + other);
   211	                    }
   212	                }
   213	            }
   214	        }
   215	        return SymbolReference.solved(winningCandidate);
   216	    }

   246	            String name, List<ResolvedType> argumentsTypes, boolean staticOnly) {
   247	
   248	        // Begin by locating methods declared "here"
   249	        List<ResolvedMethodDeclaration> candidateMethods = typeDeclaration.getDeclaredMethods().stream()
   250	                .filter(m -> m.getName().equals(name))
   251	                .filter(m -> !staticOnly || m.isStatic())
   252	                .collect(Collectors.toList());

   254	        // Next, consider methods declared within ancestors.
   255	        // Note that we only consider ancestors when we are not currently at java.lang.Object (avoiding infinite
   256	        // recursion).
   257	        if (!typeDeclaration.isJavaLangObject()) {
   258	            for (ResolvedReferenceType ancestor : typeDeclaration.getAncestors(true)) {
   259	                Optional<ResolvedReferenceTypeDeclaration> ancestorTypeDeclaration = ancestor.getTypeDeclaration();
   260	
   261	                // Avoid recursion on self
   262	                if (ancestor.getTypeDeclaration().isPresent() && typeDeclaration != ancestorTypeDeclaration.get()) {
   263	                    // Consider methods declared on self
   264	                    candidateMethods.addAll(ancestor.getAllMethodsVisibleToInheritors().stream()
   265	                            .filter(m -> m.getName().equals(name))
   266	                            .collect(Collectors.toList()));
   267	
   268	                    // consider methods from superclasses and only default methods from interfaces :
   269	                    // not true, we should keep abstract as a valid candidate
   270	                    // abstract are removed in MethodResolutionLogic.isApplicable is necessary
   271	                    SymbolReference<ResolvedMethodDeclaration> res = MethodResolutionLogic.solveMethodInType(
   272	                            ancestorTypeDeclaration.get(), name, argumentsTypes, staticOnly);
   273	                    if (res.isSolved()) {
   274	                        candidateMethods.add(res.getCorrespondingDeclaration());
   275	                    }
   276	                }
   277	            }
   278	        }
   279	
   280	        // If we haven't located any candidates that are declared on this type or its ancestors, consider the parent
   281	        // context.
   282	        // This is relevant e.g. with nested classes.
   283	        // Note that we want to avoid infinite recursion when a class is using its own method - see issue #75
   284	        // We also want to avoid infinite recursion when handling static imports - see issue #4358
   285	        if (candidateMethods.isEmpty() && !staticOnly) {
   286	            SymbolReference<ResolvedMethodDeclaration> parentSolution = context.getParent()
   287	                    .orElseThrow(() -> new RuntimeException("Parent context unexpectedly empty."))
   288	                    .solveMethod(name, argumentsTypes, staticOnly);
   289	            if (parentSolution.isSolved()) {
   290	                candidateMethods.add(parentSolution.getCorrespondingDeclaration());
   291	            }
   292	        }
   293	
   294	        // if is interface and candidate method list is empty, we should check the Object Methods
   295	        if (candidateMethods.isEmpty() && typeDeclaration.isInterface()) {
   296	            SymbolReference<ResolvedMethodDeclaration> res = MethodResolutionLogic.solveMethodInType(
   297	                    typeSolver.getSolvedJavaLangObject(), name, argumentsTypes, false);
   298	            if (res.isSolved()) {
   299	                candidateMethods.add(res.getCorrespondingDeclaration());
   300	            }
   301	        }
   302	
   303	        return MethodResolutionLogic.findMostApplicable(candidateMethods, name, argumentsTypes, typeSolver);
   304	    }
